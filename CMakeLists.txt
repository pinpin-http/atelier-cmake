cmake_minimum_required(VERSION 3.10)
project(calculator C)
add_subdirectory(include)
add_subdirectory(src)
include_directories(include)
enable_testing()

add_test(t1 src/calculator add 2 3)
set_tests_properties(t1 PROPERTIES PASS_REGULAR_EXPRESSION "^5\\.000000[\r\n]*$")
add_test(t2 src/calculator sub 3 -2)
set_tests_properties(t2 PROPERTIES PASS_REGULAR_EXPRESSION "^5\\.000000[\r\n]*$")
add_test(t3 src/calculator mul 5 5)
set_tests_properties(t3 PROPERTIES PASS_REGULAR_EXPRESSION "^25\\.000000[\r\n]*$")
add_test(t4 src/calculator div 1 5)
set_tests_properties(t4 PROPERTIES PASS_REGULAR_EXPRESSION "^0\\.200000[\r\n]*$")
add_test(car_positif src/calculator car 5)
set_tests_properties(car_positif PROPERTIES PASS_REGULAR_EXPRESSION "^25\\.000000[\r\n]*$")
add_test(car_negatif src/calculator car -5)
set_tests_properties(car_negatif PROPERTIES PASS_REGULAR_EXPRESSION "^25\\.000000[\r\n]*$")
add_test(car_zero src/calculator car 0)
set_tests_properties(car_zero PROPERTIES PASS_REGULAR_EXPRESSION "^0\\.000000[\r\n]*$")
add_test(car_decimal src/calculator car 2.5)
set_tests_properties(car_decimal PROPERTIES PASS_REGULAR_EXPRESSION "^6\\.250000[\r\n]*$")
